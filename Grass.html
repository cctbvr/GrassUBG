/* PROXY SEARCH LAUNCHER */ function openProxy(url) { if (window.grassProxy) { window.grassProxy.openInProxy(url); } else { // Fallback if proxy not initialized openGame('Web Proxy', url); } } const proxyInput = document.getElementById('main-proxy-input'); if (proxyInput) { proxyInput.addEventListener('keydown', function(e) { if(e.key === 'Enter') { let val = this.value.trim(); if(val !== '') { // Standardize URL input if(!val.startsWith('http://') && !val.startsWith('https://')) { val = 'https://' + val; } openProxy(val); this.value = ''; } } }); }
